    \begin{algorithm}
    \caption{Quicksort}
    \begin{algorithmic}
      \Procedure{Quicksort}{$A, p, r$}
        \If{$p < r$}
          \State $q \gets $ \Call{Partition}{$A, p, r$}
          \State \Call{Quicksort}{$A, p, q - 1$}
          \State \Call{Quicksort}{$A, q + 1, r$}
        \EndIf
      \EndProcedure
      \Procedure{Partition}{$A, p, r$}
        \State $x \gets A[r]$
        \State $i \gets p - 1$
        \For{$j \gets p$ \To $r - 1$}
          \If{$A[j] < x$}
            \State $i \gets i + 1$
            \State exchange
            $A[i]$ with $A[j]$
          \EndIf
        \State exchange $A[i]$ with $A[r]$
        \EndFor
      \EndProcedure
      \end{algorithmic}
    \end{algorithm}

```pseudo

\begin{algorithm}[htbp]

\caption{稀疏特征拓扑锚点识别算法}

\label{alg:anchor_topology}

\begin{algorithmic}[1]

\REQUIRE 页面截图 $I$，OCR 节点集合 $O$，图标候选集合 $C$

\ENSURE 主锚点集合 $A_c$，辅锚点集合 $A_a$，拓扑图 $G$

\STATE 提取文本节点与图标节点，合并为候选集合 $N$

\FORALL{候选节点 $n_i \in N$}

    \STATE 计算出现频率 $F_i$

    \STATE 计算文本一致性 $T_i$

    \STATE 计算位置稳定性 $P_i$

    \STATE 计算稳定度 $S_i=\alpha F_i+\beta T_i+\gamma P_i$

    \IF{$S_i \geq \tau_c$}

        \STATE 将 $n_i$ 加入主锚点集合 $A_c$

    \ELSIF{$\tau_a \leq S_i < \tau_c$}

        \STATE 将 $n_i$ 加入辅锚点集合 $A_a$

    \ENDIF

\ENDFOR

\STATE 以 $A_c$ 为主骨架，连接相邻主锚点形成初始边集合 $E_c$

\STATE 将 $A_a$ 按最近邻原则挂接至对应主锚点，形成局部修正边集合 $E_a$

\STATE 输出拓扑图 $G=(A_c\cup A_a,\;E_c\cup E_a)$

\end{algorithmic}

\end{algorithm}
